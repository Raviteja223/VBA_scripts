def inventory_age_days(
    status: pd.Series,
    event_date: pd.Series,
    settlement_date: pd.Series,
    as_of,
) -> pd.Series:
    """
    Calculate inventory age in days.

    Existing business behaviour is preserved:

    - If status == "Solde":
        use settlement_date - event_date

    - If status is missing:
        return NA

    - Otherwise:
        use reporting date - event_date

    The explicit boolean masks are required because pandas nullable
    strings can produce pd.NA from comparisons, which numpy.where()
    cannot use directly as a boolean condition.
    """

    status = status.astype("string")

    event = pd.to_datetime(
        event_date,
        errors="coerce",
    )

    settlement = pd.to_datetime(
        settlement_date,
        errors="coerce",
    )

    as_of_date = pd.Timestamp(as_of)

    # Important:
    # status.eq("Solde") can produce <NA>.
    # Convert it into a strict True/False numpy mask.
    is_settled = (
        status
        .eq("Solde")
        .fillna(False)
        .to_numpy(dtype=bool)
    )

    is_status_missing = (
        status
        .isna()
        .to_numpy(dtype=bool)
    )

    settled_age = (
        settlement - event
    ).dt.days

    active_age = (
        as_of_date - event
    ).dt.days

    result = np.where(
        is_status_missing,
        np.nan,
        np.where(
            is_settled,
            settled_age,
            active_age,
        ),
    )

    return pd.Series(
        result,
        index=status.index,
        dtype="Float64",
    )
